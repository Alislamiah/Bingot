function renderCard(f){const card=document.createElement('article');card.className='file-card'+(f.kind==='folder'?' file-folder':'');const preview=document.createElement('div');preview.className='file-preview';preview.textContent=emoji(f);if(f.type?.startsWith('image/')&&f.blob){const img=document.createElement('img');img.alt='';img.src=URL.createObjectURL(f.blob);preview.textContent='';preview.append(img);img.onload=()=>URL.revokeObjectURL(img.src)}card.append(preview);const name=document.createElement('div');name.className='file-name';name.title=f.name;name.textContent=f.name;card.append(name);const meta=document.createElement('div');meta.className='file-meta';meta.textContent=f.kind==='folder'?`${t('folder')} · ${new Date(f.createdAt).toLocaleDateString()}`:`${fmtBytes(f.size||0)} · ${new Date(f.createdAt).toLocaleDateString()}`;card.append(meta);const actions=document.createElement('div');actions.className='file-actions';if(f.kind!=='folder'){const dl=document.createElement('button');dl.className='btn';dl.textContent='↓';dl.title=t('download');dl.onclick=()=>downloadFile(f);actions.append(dl)}const star=document.createElement('button');star.className='btn';star.textContent=f.starred?'★':'☆';star.title=f.starred?t('unfavorite'):t('favorite');star.onclick=()=>toggleStar(f);actions.append(star);const del=document.createElement('button');del.className='btn ghost';del.textContent='⋯';del.title=t('delete');del.onclick=()=>showItemMenu(f);actions.append(del);card.append(actions);return card}

async function uploadFiles(list){for(const file of [...list]){const current=usedBytes();if(current+file.size>DEFAULT_QUOTA){toast(t('quotaExceeded')+' ('+file.name+')',true);continue}try{const rec={id:crypto.randomUUID?crypto.randomUUID():String(Date.now())+Math.random(),name:file.name,size:file.size,type:file.type||'',kind:'file',blob:file,createdAt:Date.now(),starred:false};await dbReq('readwrite',s=>s.put(rec));files.unshift(rec);toast(t('uploadSuccess'))}catch(e){toast(t('uploadError'),true)}}files.sort((a,b)=>b.createdAt-a.createdAt);$('#filePicker').value='';renderAll()}
